
    ```
    sips -s format png --resampleHeightWidthMax 1600 mars-image.png --out src/assets/bodies/mars-1600.png
    sips -s format png --resampleHeightWidthMax 1600 moon-image.png --out src/assets/bodies/moon-1600.png
    ```
